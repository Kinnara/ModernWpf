private void OpenButton_Click(object sender, RoutedEventArgs e)
{
    var dialog = new Microsoft.Win32.OpenFileDialog
    {
        DefaultExt = ".rtf",
        Filter = "Rich Text (*.rtf)|*.rtf|All files (*.*)|*.*"
    };

    if (dialog.ShowDialog() == true)
    {
        var range = new TextRange(editor.Document.ContentStart, editor.Document.ContentEnd);
        using (var stream = System.IO.File.OpenRead(dialog.FileName))
        {
            range.Load(stream, DataFormats.Rtf);
        }
    }
}

private void SaveButton_Click(object sender, RoutedEventArgs e)
{
    var dialog = new Microsoft.Win32.SaveFileDialog
    {
        DefaultExt = ".rtf",
        FileName = "New Document",
        Filter = "Rich Text (*.rtf)|*.rtf"
    };

    if (dialog.ShowDialog() == true)
    {
        var range = new TextRange(editor.Document.ContentStart, editor.Document.ContentEnd);
        using (var stream = System.IO.File.Create(dialog.FileName))
        {
            range.Save(stream, DataFormats.Rtf);
        }
    }
}

private void BoldButton_Click(object sender, RoutedEventArgs e)
{
    EditingCommands.ToggleBold.Execute(null, editor);
    editor.Focus();
}

private void ItalicButton_Click(object sender, RoutedEventArgs e)
{
    EditingCommands.ToggleItalic.Execute(null, editor);
    editor.Focus();
}

private void ColorButton_Click(object sender, RoutedEventArgs e)
{
    if (sender is Button clickedColor &&
        clickedColor.Content is Rectangle rectangle &&
        rectangle.Fill is SolidColorBrush brush)
    {
        editor.Selection.ApplyPropertyValue(TextElement.ForegroundProperty, brush);
    }

    fontColorButton.Flyout.Hide();
    editor.Focus();
}

private void FindBoxHighlightMatches(object sender, RoutedEventArgs e)
{
    if (string.IsNullOrEmpty(findBox.Text))
    {
        return;
    }

    var range = new TextRange(editor.Document.ContentStart, editor.Document.ContentEnd);
    if (range.Text.IndexOf(findBox.Text, StringComparison.CurrentCultureIgnoreCase) >= 0)
    {
        editor.Focus();
    }
}

private void FindBoxRemoveHighlights(object sender, RoutedEventArgs e)
{
    editor.Selection.ApplyPropertyValue(TextElement.BackgroundProperty, DependencyProperty.UnsetValue);
    editor.Selection.ApplyPropertyValue(TextElement.ForegroundProperty, DependencyProperty.UnsetValue);
}

private void Editor_GotFocus(object sender, RoutedEventArgs e)
{
    editor.Selection.ApplyPropertyValue(TextElement.BackgroundProperty, DependencyProperty.UnsetValue);
}
